/*/{Protheus.doc} AP_UTF8
Acento UTF-8 que chega quebrado em 2 caracteres CP1252 (igual ao CH_DECODESAFE do CNSA001).
/*/
Static Function AP_UTF8(cTexto)
    Local cRet := ""
    Local nLen := Len(cTexto)
    Local nI   := 1
    Local nB1  := 0
    Local nB2  := 0

    While nI <= nLen
        nB1 := Asc(SubStr(cTexto, nI, 1))
        If (nB1 == 195 .Or. nB1 == 194) .And. nI < nLen
            nB2 := Asc(SubStr(cTexto, nI + 1, 1))
            If nB2 >= 128 .And. nB2 <= 191
                cRet += Chr(((nB1 - 192) * 64) + (nB2 - 128))
                nI   += 2
                Loop
            EndIf
        EndIf
        cRet += SubStr(cTexto, nI, 1)
        nI++
    EndDo

Return cRet
