/*/{Protheus.doc} CH_PROXIMASEQ
Proxima Z6_SEQ do tecnico na data.
@param  cTecnico   Codigo do tecnico (AA1)
@param  dData      Data do agendamento
@return Sequencia formatada em 3 digitos
/*/
Static Function CH_PROXIMASEQ(cTecnico, dData)
    Local cAlias := GetNextAlias()
    Local nSeq   := 1

    BeginSql Alias cAlias
        SELECT MAX(SZ6.Z6_SEQ) MAXSEQ
        FROM %Table:SZ6% SZ6
        WHERE SZ6.Z6_FILIAL  = %xFilial:SZ6%
          AND SZ6.Z6_TECNICO = %Exp:cTecnico%
          AND SZ6.Z6_DTAGE   = %Exp:dData%
          AND %NotDel%
    EndSql

    If !(cAlias)->(Eof()) .And. !Empty((cAlias)->MAXSEQ)
        nSeq := Val((cAlias)->MAXSEQ) + 1
    EndIf
    (cAlias)->(DbCloseArea())

Return PadL(AllTrim(Str(nSeq)), 3, "0")

/*/{Protheus.doc} CH_GRAVARAGENDA
Grava um dia de agenda via Model (CNSA003) em vez de RecLock direto na SZ6. LockByName/UnLockByName: assinatura pode variar por build do TOTVS.
@param  cIdCh              Id do chamado (ZA1_IDCH)
@param  cDataIni           Data inicial do intervalo (AAAA-MM-DD)
@param  cDataFim           Data final do intervalo (AAAA-MM-DD, default = cDataIni)
@param  cCodTecnico        Codigo do tecnico (AA1)
@param  cHoraIni           Hora inicial (HH:MM)
@param  cHoraFim           Hora final (HH:MM)
@param  cCodCliente        Codigo do cliente (SA1)
@param  cLojaCliente       Loja do cliente (default "01")
@param  cDescricao         Descricao/servico do agendamento
@param  cConfirmacao       Confirmacao do agendamento
@param  cTurno             Turno do agendamento
@param  cAgInterna         Agenda interna (SIM/NAO)
@param  cAgCobravel        Agenda cobravel (SIM/NAO)
@param  cTipoAgenda        Tipo de agenda
@param  cProjeto           Codigo do projeto
@param  cRevisao           Revisao do projeto
@param  cTarefa            Codigo da tarefa
@param  cTipoHora          Tipo de hora
@param  cCodAgendador      Codigo de quem esta agendando (default: usuario da sessao)
@param  lNotificarTecnico  .T. pra enviar ao analista o e-mail de agendamento (default .T.)
@return Array {nStatusHTTP, cCorpoJson}
/*/
Static Function CH_GRAVARAGENDA(cIdCh, cDataIni, cDataFim, cCodTecnico, cHoraIni, cHoraFim, cCodCliente, ;
    cLojaCliente, cDescricao, cConfirmacao, cTurno, cAgInterna, cAgCobravel, cTipoAgenda, cProjeto, ;
    cRevisao, cTarefa, cTipoHora, cCodAgendador, lNotificarTecnico)
    Local nCham       := 0
    Local dDataIni    := CToD("")
    Local dDataFim    := CToD("")
    Local dAtual      := CToD("")
    Local cDataIso    := ""
    Local nCriados    := 0
    Local nTotalDias  := 0
    Local cSeq        := ""
    Local oJson       := JsonObject():New()
    Local oErro       := Nil
    Local oErroLoop   := Nil
    Local oErroDia    := Nil
    Local cLockKey    := ""
    Local lLockOk     := .F.
    Local aFalhas     := {}
    Local oFalha      := Nil
    Local nTentativa  := 0
    Local lGravouDia  := .F.
    Local oModel      := Nil
    Local oModelSZ6   := Nil
    Local lOkModel    := .F.

    Begin Sequence

        // Obrigatorios ficam no front; aqui so protecoes tecnicas (chamado inexistente, data invalida, teto de 730 dias).
        If Empty(cDataFim)
            cDataFim := cDataIni
        EndIf
        If Empty(cLojaCliente) .And. !Empty(cCodCliente)
            cLojaCliente := "01"
        EndIf
        Default lNotificarTecnico := .T.

        If !CH_ACHARZA1(cIdCh)
            Return {404, CH_ERRO("Chamado nao encontrado.")}
        EndIf

        nCham := Val(cIdCh)

        dDataIni := STOD(StrTran(cDataIni, "-", ""))
        dDataFim := STOD(StrTran(cDataFim, "-", ""))

        If Empty(dDataIni) .Or. Empty(dDataFim)
            Return {400, CH_ERRO("Data invalida.")}
        EndIf

        If (dDataFim - dDataIni) > 730
            Return {400, CH_ERRO("Intervalo maximo de 2 anos (730 dias) por agendamento.")}
        EndIf

        If Empty(cCodAgendador)
            cCodAgendador := AllTrim(Posicione("AA1", 4, xFilial("AA1") + __cUserId, "AA1_NOMTEC"))
        EndIf

        nTotalDias := (dDataFim - dDataIni) + 1

        cLockKey := "CNS001_AGENDA_" + cCodTecnico
        lLockOk  := LockByName(cLockKey, .F., .T., 10)
        If !lLockOk
            Return {423, CH_ERRO("Outro agendamento para este tecnico esta em andamento. Tente novamente.")}
        EndIf

        Begin Sequence

            dAtual := dDataIni
            While dAtual <= dDataFim
                cDataIso := SubStr(DTOS(dAtual), 1, 4) + "-" + SubStr(DTOS(dAtual), 5, 2) + "-" + SubStr(DTOS(dAtual), 7, 2)

                lGravouDia := .F.
                oErroDia   := Nil
                For nTentativa := 1 To 3
                    Begin Sequence
                        cSeq := CH_PROXIMASEQ(cCodTecnico, dAtual)

                        oModel := FWLoadModel("CNSA003")
                        oModel:SetOperation(MODEL_OPERATION_INSERT)
                        oModel:Activate()
                        oModelSZ6 := oModel:GetModel("ModelSZ6_Main")

                        oModelSZ6:SetValue("Z6_FILIAL",  xFilial("SZ6"))
                        oModelSZ6:SetValue("Z6_CHAMADO", nCham)
                        oModelSZ6:SetValue("Z6_DTAGE",   dAtual)
                        oModelSZ6:SetValue("Z6_TECNICO", cCodTecnico)
                        oModelSZ6:SetValue("Z6_SEQ",     cSeq)
                        oModelSZ6:SetValue("Z6_HMINI",   cHoraIni)
                        oModelSZ6:SetValue("Z6_HMFIM",   cHoraFim)
                        oModelSZ6:SetValue("Z6_CLIENTE", cCodCliente)
                        oModelSZ6:SetValue("Z6_LOJA",    cLojaCliente)
                        oModelSZ6:SetValue("Z6_SERVICO", cDescricao)
                        oModelSZ6:SetValue("Z6_CONFIRM", If(Empty(cConfirmacao), "NAO", cConfirmacao))
                        oModelSZ6:SetValue("Z6_TURNO",   Val(cTurno))
                        oModelSZ6:SetValue("Z6_INTERNO", If(Empty(cAgInterna), "NAO", cAgInterna))
                        oModelSZ6:SetValue("Z6_COBRAR",  If(Empty(cAgCobravel), "NAO", cAgCobravel))
                        oModelSZ6:SetValue("Z6_TIPOAG",  cTipoAgenda)
                        oModelSZ6:SetValue("Z6_PROJET",  cProjeto)
                        oModelSZ6:SetValue("Z6_REVISA",  cRevisao)
                        oModelSZ6:SetValue("Z6_TAREFA",  cTarefa)
                        oModelSZ6:SetValue("Z6_TIPO_HR", cTipoHora)
                        oModelSZ6:SetValue("Z6_QGRAVOU", cCodAgendador)

                        lOkModel := oModel:VldData() .And. oModel:CommitData()
                        oModel:DeActivate(lOkModel)

                        lGravouDia := lOkModel
                        If !lGravouDia
                            oErroDia := ErrorNew()
                            oErroDia:Description := "Falha de validacao/gravacao do Model CNSA003"
                        EndIf
                    Recover Using oErroDia
                        lGravouDia := .F.
                        // Excecao tambem desativa o model (senao fica pendurado entre tentativas).
                        If oModel <> Nil
                            oModel:DeActivate()
                        EndIf
                    End Sequence

                    If lGravouDia
                        Exit
                    EndIf
                Next nTentativa

                If lGravouDia
                    nCriados++
                Else
                    oFalha := JsonObject():New()
                    oFalha['data'] := cDataIso
                    oFalha['erro'] := "Falha ao gravar apos 3 tentativas: " + If(oErroDia != Nil, oErroDia:Description, "erro desconhecido")
                    AAdd(aFalhas, oFalha)
                EndIf

                dAtual := dAtual + 1
            EndDo

        Recover Using oErroLoop
        End Sequence

        UnLockByName(cLockKey, .T.)

        If oErroLoop != Nil
            Return {500, CH_ERRO(oErroLoop:Description)}
        EndIf

        // E-mail ao tecnico sai do evento da Agenda (CNS003OUTL), um por agendamento.

        oJson['sucesso']    := nCriados > 0
        oJson['idCh']       := cIdCh
        oJson['quantidade'] := nCriados
        oJson['falhas']     := aFalhas
        oJson['mensagem']   := AllTrim(Str(nCriados)) + " de " + AllTrim(Str(nTotalDias)) + " dia(s) criado(s) com sucesso."

        Return {If(nCriados > 0, 200, 500), oJson:ToJson()}

    Recover Using oErro
        Return {500, CH_ERRO(oErro:Description)}
    End Sequence

Return {500, CH_ERRO("Erro desconhecido.")}

/*/{Protheus.doc} CH_AGENDAJSON
JSON das agendas (SZ6) do chamado, por data/hora.
@param  cIdCh   Id do chamado (ZA1_IDCH)
@return Array {nStatusHTTP, cCorpoJson}
/*/
Static Function CH_AGENDAJSON(cIdCh)
    Local cAlias   := ""
    Local oJson    := JsonObject():New()
    Local aItens   := {}
    Local oLinha   := Nil
    Local cDataIso := ""
    Local cCodCli  := ""
    Local cLoja    := ""
    Local oErro    := Nil
    // Z6_CHAMADO e numerico: comparar com numero (string nunca bate no Oracle).
    Local nCham    := Val(cIdCh)

    Begin Sequence

        If !CH_ACHARZA1(cIdCh)
            Return {404, CH_ERRO("Chamado nao encontrado.")}
        EndIf

        cAlias := GetNextAlias()
        BeginSql Alias cAlias
            SELECT Z6_DTAGE, Z6_HMINI, Z6_HMFIM, Z6_TECNICO, Z6_CLIENTE, Z6_LOJA, Z6_SERVICO
            FROM %Table:SZ6% SZ6
            WHERE Z6_FILIAL  = %xFilial:SZ6%
              AND Z6_CHAMADO = %Exp:nCham%
              AND %NotDel%
            ORDER BY Z6_DTAGE, Z6_HMINI
        EndSql

        While !(cAlias)->(Eof())
            If ValType((cAlias)->Z6_DTAGE) == "D"
                cDataIso := DToS((cAlias)->Z6_DTAGE)
            Else
                cDataIso := AllTrim((cAlias)->Z6_DTAGE)
            EndIf
            If Len(cDataIso) == 8
                cDataIso := SubStr(cDataIso, 1, 4) + "-" + SubStr(cDataIso, 5, 2) + "-" + SubStr(cDataIso, 7, 2)
            EndIf

            cCodCli := AllTrim((cAlias)->Z6_CLIENTE)
            cLoja   := AllTrim((cAlias)->Z6_LOJA)

            oLinha := JsonObject():New()
            oLinha['data']       := cDataIso
            oLinha['horaInicio'] := AllTrim((cAlias)->Z6_HMINI)
            oLinha['horaFim']    := AllTrim((cAlias)->Z6_HMFIM)
            oLinha['tecnico']    := AllTrim(Posicione("AA1", 1, xFilial("AA1") + AllTrim((cAlias)->Z6_TECNICO), "AA1_NOMTEC"))
            oLinha['cliente']    := AllTrim(Posicione("SA1", 1, xFilial("SA1") + cCodCli + cLoja, "A1_NREDUZ"))
            oLinha['descricao']  := CH_DECODESAFE(AllTrim((cAlias)->Z6_SERVICO))
            AAdd(aItens, oLinha)

            (cAlias)->(DbSkip())
        EndDo
        (cAlias)->(DbCloseArea())

        oJson['idCh']  := cIdCh
        oJson['items'] := aItens

        Return {200, oJson:ToJson()}

    Recover Using oErro
        Return {500, CH_ERRO(oErro:Description)}
    End Sequence

Return {500, CH_ERRO("Erro desconhecido.")}

/*/{Protheus.doc} CH_DECODESAFE
Acento UTF-8 que chega quebrado em 2 caracteres CP1252: remonta o par de bytes. Nunca da erro.
@param  cTexto   Texto potencialmente com UTF-8 mal interpretado como CP1252
@return cRet   Texto com os pares de bytes UTF-8 decodificados
/*/
Static Function CH_DECODESAFE(cTexto)
    Local cRet  := ""
    Local nLen  := Len(cTexto)
    Local nI    := 1
    Local nB1, nB2

    While nI <= nLen
        nB1 := Asc(SubStr(cTexto, nI, 1))
        If (nB1 == 195 .Or. nB1 == 194) .And. nI < nLen
            nB2 := Asc(SubStr(cTexto, nI + 1, 1))
            If nB2 >= 128 .And. nB2 <= 191
                cRet += Chr(((nB1 - 192) * 64) + (nB2 - 128))
                nI   += 2
                Loop
            EndIf
        EndIf
        cRet += SubStr(cTexto, nI, 1)
        nI++
    EndDo

Return cRet

/*/{Protheus.doc} CH_BACKFILLNOMES
Execucao unica: preenche ZA1_NOME/ZA1_NOMTEC dos chamados antigos (campos que eram virtuais).
Pode rodar de novo: so grava o que estiver diferente.
/*/
User Function CH_BACKFILLNOMES()
    Local cCli      := ""
    Local cLoja     := ""
    Local cCons     := ""
    Local cNomeCli  := ""
    Local cNomeTec  := ""

    dbSelectArea("ZA1")
    ZA1->(DbGoTop())
    While !ZA1->(Eof())
        cCli  := AllTrim(ZA1->ZA1_CLIENT)
        cLoja := AllTrim(ZA1->ZA1_LOJA)
        cCons := AllTrim(ZA1->ZA1_CONSUL)

        cNomeCli := If(Empty(cCli), "", AllTrim(Posicione("SA1", 1, xFilial("SA1") + cCli + cLoja, "A1_NREDUZ")))
        cNomeTec := If(Empty(cCons), "", AllTrim(Posicione("AA1", 1, xFilial("AA1") + cCons, "AA1_NOMTEC")))

        If AllTrim(ZA1->ZA1_NOME) != cNomeCli .Or. AllTrim(ZA1->ZA1_NOMTEC) != cNomeTec
            RecLock("ZA1", .F.)
            ZA1->ZA1_NOME   := cNomeCli
            ZA1->ZA1_NOMTEC := cNomeTec
            ZA1->(MsUnlock())
        EndIf

        ZA1->(DbSkip())
    EndDo

Return

/*/{Protheus.doc} CH_TOKENGRAPH
Token do Microsoft Graph (OAuth2 client credentials: MV_CNSATEN/MV_CNSACLI/MV_CNSASEC).
@return cToken   Access token do Graph, ou "" em falha/configuracao ausente
/*/
Static Function CH_TOKENGRAPH()
    Local cTenant := AllTrim(GETMV("MV_CNSATEN"))
    Local cClient := AllTrim(GETMV("MV_CNSACLI"))
    Local cSecret := AllTrim(GETMV("MV_CNSASEC"))
    Local oRest, oJson, cBody, cToken := ""

    If Empty(cTenant) .Or. Empty(cClient) .Or. Empty(cSecret)
        ConOut("CH_TOKENGRAPH: MV_CNSATEN/MV_CNSACLI/MV_CNSASEC nao configurados no Configurador - e-mail nao sera enviado.")
        Return ""
    EndIf

    cBody := "grant_type=client_credentials"
    cBody += "&client_id="     + CH_CODIFICARURL(cClient)
    cBody += "&client_secret=" + CH_CODIFICARURL(cSecret)
    cBody += "&scope="         + CH_CODIFICARURL("https://graph.microsoft.com/.default")

    oRest := FWRest():New("https://login.microsoftonline.com")
    oRest:setPath("/" + cTenant + "/oauth2/v2.0/token")
    oRest:SetPostParams(cBody)
    If !oRest:Post({"Content-Type: application/x-www-form-urlencoded"})
        ConOut("CH_TOKENGRAPH: falha no POST de token (Azure AD): " + oRest:GetLastError())
        FreeObj(oRest)
        Return ""
    EndIf

    oJson := JsonObject():New()
    If Empty(oJson:FromJson(oRest:GetResult())) .And. ValType(oJson["access_token"]) == "C"
        cToken := oJson["access_token"]
    Else
        ConOut("CH_TOKENGRAPH: resposta do Azure AD sem access_token: " + oRest:GetResult())
    EndIf
    FreeObj(oJson)
    FreeObj(oRest)
Return cToken

/*/{Protheus.doc} CH_CODIFICARURL
Codifica um texto em percent-encoding (application/x-www-form-urlencoded),
usado para montar o corpo da requisicao de token do Azure AD.
@param  cTexto   Texto a codificar
@return cRet   Texto codificado em URL
/*/
Static Function CH_CODIFICARURL(cTexto)
    Local cRet := ""
    Local nI   := 0
    Local cCh  := ""

    For nI := 1 To Len(cTexto)
        cCh := SubStr(cTexto, nI, 1)
        If (cCh >= "A" .And. cCh <= "Z") .Or. ;
           (cCh >= "a" .And. cCh <= "z") .Or. ;
           (cCh >= "0" .And. cCh <= "9") .Or. ;
           (cCh $ "-_.~")
            cRet += cCh
        Else
            cRet += "%" + CH_HEXDOISDIG(Asc(cCh))
        EndIf
    Next nI
Return cRet

/*/{Protheus.doc} CH_HEXDOISDIG
Converte um valor numerico (0-255) para 2 digitos hexadecimais, usado
pela codificacao percent-encoding (CH_CODIFICARURL).
@param  nVal   Valor numerico a converter
@return String hexadecimal de 2 digitos
/*/
Static Function CH_HEXDOISDIG(nVal)
    Local cHex := "0123456789ABCDEF"
Return SubStr(cHex, Int(nVal/16)+1, 1) + SubStr(cHex, (nVal % 16)+1, 1)

/*/{Protheus.doc} CH_NOTIFICARDESIGNA
E-mail de designacao ao tecnico (e ao cliente, se pedido) com os dados do chamado e da agenda.
@param  cIdCh                 Id do chamado (ZA1_IDCH)

@return .T./.F. indicando sucesso no envio ao tecnico (via CH_ENVIAREMAIL)
/*/

    Local cNomeTec   := ""
    Local cEmailTec  := ""
    Local cNomeCli   := ""
    Local cEmailCli  := ""
    Local cAssunto   := ""
    Local cDataAge   := DToC(Date())
    Local cHmini     := "00:00"
    Local cHmfim     := "00:00"
    Local cNomeDe    := ""
    Local cCorpo     := ""
    Local cCorpoCli  := ""
    Local cDescricao := ""
    Local lOk        := .F.
    Local cAlias     := GetNextAlias()
    Local nCham      := Val(cIdCh)

    Default lNotificarCliente := .F.
    Default lNotificarTecnico := .T.

    If Empty(cCodConsultorNovo)
        Return .F.
    EndIf

    cNomeTec  := AllTrim(Posicione("AA1", 1, xFilial("AA1") + cCodConsultorNovo, "AA1_NOMTEC"))
    cEmailTec := AllTrim(Posicione("AA1", 1, xFilial("AA1") + cCodConsultorNovo, "AA1_EMAIL"))
    // "Designado por" = quem executa a acao (usuario logado), nao o tecnico.
    cNomeDe   := AllTrim(Posicione("AA1", 4, xFilial("AA1") + __cUserId, "AA1_NOMTEC"))

    If Empty(cEmailTec)
        ConOut("CH_NOTIFICARDESIGNA: tecnico " + cCodConsultorNovo + " sem AA1_EMAIL cadastrado - e-mail de designacao nao enviado ao tecnico.")
    EndIf

    If CH_ACHARZA1(cIdCh)
        cAssunto := AllTrim(ZA1->ZA1_ASSUNT)
        cEmailCli := AllTrim(ZA1->ZA1_EMAIL)
        cDescricao := AllTrim(ZA1->ZA1_ANOT)
        cNomeCli := AllTrim(Posicione("SA1", 1, xFilial("SA1") + AllTrim(ZA1->ZA1_CLIENT) + AllTrim(ZA1->ZA1_LOJA), "A1_NREDUZ"))
        If ValType(ZA1->ZA1_DATA) == "D" .And. !Empty(ZA1->ZA1_DATA)
            cDataAge := DToC(ZA1->ZA1_DATA)
        ElseIf ValType(ZA1->ZA1_DATA) == "C" .And. Len(AllTrim(ZA1->ZA1_DATA)) == 8
            cDataAge := DToC(STOD(AllTrim(ZA1->ZA1_DATA)))
        EndIf
    EndIf

    // Horario do agendamento (SZ6) real do chamado.
    BeginSql Alias cAlias
        SELECT Z6_DTAGE, Z6_HMINI, Z6_HMFIM
        FROM %Table:SZ6% SZ6
        WHERE Z6_FILIAL  = %xFilial:SZ6%
          AND Z6_CHAMADO = %Exp:nCham%
          AND %NotDel%
    EndSql
    If !(cAlias)->(Eof())
        If ValType((cAlias)->Z6_DTAGE) == "D"
            cDataAge := DToC((cAlias)->Z6_DTAGE)
        Else
            cDataAge := AllTrim((cAlias)->Z6_DTAGE)
            If Len(cDataAge) == 8
                cDataAge := SubStr(cDataAge,7,2)+"/"+SubStr(cDataAge,5,2)+"/"+SubStr(cDataAge,1,4)
            EndIf
        EndIf
        cHmini := AllTrim((cAlias)->Z6_HMINI)
        cHmfim := AllTrim((cAlias)->Z6_HMFIM)
    EndIf
    (cAlias)->(DbCloseArea())

    If Empty(cHmini) ; cHmini := "00:00" ; EndIf
    If Empty(cHmfim) ; cHmfim := "00:00" ; EndIf

    // Cliente so recebe com ZA1_EMAIL e se pediram notificar.
    If lNotificarCliente .And. !Empty(cEmailCli)
        cCorpoCli := CH_HTMLDESIGNACLIENTE(cNomeTec, cIdCh, cAssunto, cDataAge, cHmini, cHmfim)
        CH_ENVIAREMAIL(cEmailCli, "Tecnico Designado - Chamado " + cIdCh, cCorpoCli, "apia@apia.com.br")
    EndIf

    If !lNotificarTecnico .Or. Empty(cEmailTec)
        Return .F.
    EndIf

    cCorpo := CH_HTMLDESIGNA(cNomeTec, cIdCh, cNomeCli, cAssunto, cDataAge, cHmini, cHmfim, cNomeDe, cEmailCli, cDescricao)

    lOk := CH_ENVIAREMAIL(cEmailTec, "Chamado Designado " + cIdCh, cCorpo, "apia@apia.com.br")

Return lOk

/*/{Protheus.doc} GetData
Le o model pelo REST generico, com ZA1_NOME/ZA1_NOMTEC (o gatilho nao roda na leitura).
@param  lFieldDetail    Inclui detalhe dos campos no JSON
@param  lFieldVirtual   Inclui campos virtuais no JSON
@param  lFieldEmpty     Inclui campos vazios no JSON
@param  lFirstLevel     Retorna somente o primeiro nivel do Model
@param  lInternalID     Inclui ID interno no JSON
@return cRet            JSON com os dados do Model, campos de nome ja injetados
/*/
Method GetData(lFieldDetail, lFieldVirtual, lFieldEmpty, lFirstLevel, lInternalID) Class oRestChamados
Local cRet
Local oErroExcpt := Nil
Local cCodCli    := ""
Local cLoja      := ""
Local cCodTec    := ""
Local cNomeCli   := ""
Local cNomeTec   := ""
Local jRegistro  := Nil
Local aModels    := Nil
Local aFields    := Nil
Local nX
Local nY
    self:oModel:SetOperation(MODEL_OPERATION_VIEW)
    self:oModel:Activate()
    cCodCli := self:oModel:GetValue('ZA1MASTER', 'ZA1_CLIENT')
    cLoja   := self:oModel:GetValue('ZA1MASTER', 'ZA1_LOJA')
    cCodTec := self:oModel:GetValue('ZA1MASTER', 'ZA1_CONSUL')

    cRet := self:oModel:GetJsonData(lFieldDetail,,lFieldVirtual,,lFieldEmpty,.T./*lPK*/,.T./*lPKEncoded*/,self:aFields,lFirstLevel,lInternalID)
    self:oModel:DeActivate()

    Begin Sequence
        cNomeCli := Posicione("SA1", 1, xFilial("SA1") + cCodCli + cLoja, "A1_NREDUZ")
        cNomeTec := Posicione("AA1", 1, xFilial("AA1") + cCodTec, "AA1_NOMTEC")
        jRegistro := JsonObject():New()
        jRegistro:fromJson(cRet)
        If jRegistro != Nil
            aModels := jRegistro:GetJsonObject("models")

            For nX := 1 To Len(aModels)
            If aModels[nX]:GetJsonText("id") == "ZA1MASTER"
                aFields := aModels[nX]:GetJsonObject("fields")
                For nY := 1 To Len(aFields)
                    If aFields[nY]:GetJsonText("id") == "ZA1_NOME"
                        aFields[nY]['value'] := cNomeCli
                    ElseIf aFields[nY]:GetJsonText("id") == "ZA1_NOMTEC"
                        aFields[nY]['value'] := cNomeTec
                    EndIf
                Next nY
                Exit
            EndIf
        Next nX

            cRet := jRegistro:toJson()
        EndIf
    Recover Using oErroExcpt
        ConOut("[oRestChamados:GetData] Falha ao injetar nome cliente/tecnico: " + oErroExcpt:Description)
    End Sequence
Return cRet

/*/{Protheus.doc} SaveData
Grava pelo REST generico: alteracao pelo _Super; inclusao pelo CH_NUCLEOINCLUIR.
@param  cPK      Chave primaria do registro (vazia = inclusao)
@param  cData    Corpo JSON da requisicao com os dados a gravar
@param  cError   Mensagem de erro (preenchida por referencia em caso de falha)
@return lRet     .T. se gravou com sucesso, .F. caso contrario
/*/
Method SaveData(cPK, cData, cError) Class oRestChamados
Local lRet        := .T.
Local jBody       := Nil
Local oErroExcpt  := Nil
Local aResultado  := {}

    // Alterar: generico (_Super), igual oRestAgenda do CNSA003.
    If !Empty(cPK)
        Return _Super:SaveData(cPK, cData, cError)
    EndIf

    Begin Sequence
        jBody := JsonObject():New()
        jBody:fromJson(cData)
        If jBody == Nil
            cError := "Corpo da requisicao invalido (JSON malformado)."
            lRet   := .F.
        Else
            // Incluir pelo nucleo unico (CH_NUCLEOINCLUIR).
            aResultado := CH_NUCLEOINCLUIR(;
                AllTrim(jBody:GetJsonText("ZA1_ASSUNT")), ;
                jBody:GetJsonText("ZA1_DATA"), ;
                AllTrim(jBody:GetJsonText("ZA1_TIPO")), ;
                AllTrim(jBody:GetJsonText("ZA1_CLIENT")), ;
                AllTrim(jBody:GetJsonText("ZA1_LOJA")), ;
                AllTrim(jBody:GetJsonText("ZA1_EMAIL")), ;
                AllTrim(jBody:GetJsonText("ZA1_CONSUL")), ;
                AllTrim(jBody:GetJsonText("ZA1_PROJ")), ;
                AllTrim(jBody:GetJsonText("ZA1_TAREFA")))

            lRet := (aResultado[1] == 200)
            If !lRet
                cError := aResultado[2]
            EndIf
        EndIf
    Recover Using oErroExcpt
        lRet   := .F.
        cError := "Erro interno: " + oErroExcpt:Description
        ConOut("[oRestChamados:SaveData] " + cError)
    End Sequence
Return lRet
